    const canvas = document.getElementById('pizarra'); const ctx = canvas.getContext('2d'); let dibujando = false, haDibujado = false;
    function redimensionarCanvas() {
        const tempCanvas = document.createElement('canvas');
        tempCanvas.width = canvas.width;
        tempCanvas.height = canvas.height;
        const tempCtx = tempCanvas.getContext('2d');
        if (canvas.width > 0 && canvas.height > 0) {
            tempCtx.drawImage(canvas, 0, 0);
        }

        canvas.width = canvas.parentElement.clientWidth;
        canvas.height = canvas.parentElement.clientHeight;
        ctx.lineWidth = 3;
        ctx.strokeStyle = '#162839'; // Primary color
        ctx.lineCap = 'round';
        ctx.lineJoin = 'round';

        if (tempCanvas.width > 0 && tempCanvas.height > 0) {
            ctx.drawImage(tempCanvas, 0, 0);
        }
    }
    window.addEventListener('resize', redimensionarCanvas); redimensionarCanvas();

    function getPos(e) {
        const rect = canvas.getBoundingClientRect();
        const clientX = e.touches ? e.touches[0].clientX : e.clientX;
        const clientY = e.touches ? e.touches[0].clientY : e.clientY;
        return {
            x: clientX - rect.left,
            y: clientY - rect.top
        };
    }

    function startDrawing(e) {
        e.preventDefault();
        dibujando = true;
        const pos = getPos(e);
        ctx.beginPath();
        ctx.moveTo(pos.x, pos.y);
    }

    function draw(e) {
        if (!dibujando) return;
        e.preventDefault();
        const pos = getPos(e);
        ctx.lineTo(pos.x, pos.y);
        ctx.stroke();
        haDibujado = true;
    }

    function stopDrawing() {
        dibujando = false;
    }

    canvas.addEventListener('mousedown', startDrawing);
    canvas.addEventListener('mousemove', draw);
    canvas.addEventListener('mouseup', stopDrawing);
    canvas.addEventListener('mouseout', stopDrawing);

    canvas.addEventListener('touchstart', startDrawing);
    canvas.addEventListener('touchmove', draw);
    canvas.addEventListener('touchend', stopDrawing);

    function limpiarCanvas() { ctx.clearRect(0,0,canvas.width,canvas.height); haDibujado=false; }

    function recuperarPin() {
        if(confirm("¿Enviar enlace de recuperación de PIN a tu correo?")) {
            fetch('/api/solicitar-recuperacion/', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ email: '{{ email_firmante }}' })
            }).then(() => alert("Correo enviado. Revisa tu bandeja de entrada."));
        }
    }

    function enviar(esPin) {
        let variables = {};
        let valid = true;
        document.querySelectorAll('.doc-var-input').forEach(input => {
            if(!input.value || input.value.trim() === '') {
                valid = false;
                input.classList.add('border-status-danger');
                input.classList.remove('border-outline-variant');
            } else {
                input.classList.remove('border-status-danger');
                input.classList.add('border-outline-variant');
            }
            variables[input.dataset.key] = input.value.trim();
        });

        if(!valid) return alert("Por favor, completa todos los campos obligatorios antes de continuar.");

        let payload = { variables: variables };

        if(esPin) {
            const pin = document.getElementById('inputPin').value;
            if(!pin || pin.length < 4) return alert("Ingresa un PIN de seguridad válido.");
            payload.pin = pin;
        } else {
            if(!haDibujado) return alert("Dibuja tu firma autógrafa para proceder.");
            payload.firma_base64 = canvas.toDataURL('image/png');
        }

        const btn = document.getElementById(esPin ? 'btnFirmarPin' : 'btnFirmarCanvas');
        const originalText = btn.innerHTML;
        btn.disabled = true; btn.innerHTML = '<span class="material-symbols-outlined animate-spin">sync</span> Certificando...';

        let procesarUrl = '/api/procesar/{{ token }}/';
        {% if firmante_token %}
            procesarUrl += '{{ firmante_token }}/';
        {% endif %}

        fetch(procesarUrl, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(payload)
        }).then(r => r.json()).then(data => {
            if(data.status === 'success') {
                document.querySelector('.content-body || main > div').innerHTML = `
                    <div class="text-center py-12">
                        <div class="inline-flex items-center justify-center w-20 h-20 bg-status-success/10 rounded-full mb-6">
                            <span class="material-symbols-outlined text-status-success text-5xl">check_circle</span>
                        </div>
                        <h2 class="text-3xl font-bold text-primary mb-4">¡Certificación Exitosa!</h2>
                        <p class="text-lg text-text-muted mb-8 max-w-lg mx-auto">Tu firma y los datos han sido estampados de forma segura en el documento oficial.</p>
                        <p class="text-sm text-outline">Ya puedes cerrar esta ventana con total seguridad.</p>
                    </div>`;
            }
            else {
                alert("Error: " + data.error);
                btn.disabled = false;
                btn.innerHTML = originalText;
            }
        }).catch(err => {
            alert("Ocurrió un error de conexión.");
            btn.disabled = false;
            btn.innerHTML = originalText;
        });

<script data-purpose="canvas-handling">
    const canvas = document.getElementById('pizarra');
    const ctx = canvas.getContext('2d');
    let dibujando = false;
    let haDibujado = false;

    function redimensionarCanvas() {
      // Preserve content on resize (common on mobile keyboard popup)
      const tempCanvas = document.createElement('canvas');
      tempCanvas.width = canvas.width;
      tempCanvas.height = canvas.height;
      const tempCtx = tempCanvas.getContext('2d');
      if (canvas.width > 0 && canvas.height > 0) {
        tempCtx.drawImage(canvas, 0, 0);
      }

      canvas.width = canvas.offsetWidth;
      canvas.height = canvas.offsetHeight;

      ctx.lineWidth = 2.5;
      ctx.lineCap = 'round';
      ctx.lineJoin = 'round';
      ctx.strokeStyle = '#000080'; // Navy Blue Ink

      if (tempCanvas.width > 0 && tempCanvas.height > 0) {
        ctx.drawImage(tempCanvas, 0, 0);
      }
    }

    window.addEventListener('resize', redimensionarCanvas);
    redimensionarCanvas();

    // Helper to get mouse/touch position relative to canvas
    function getPos(e) {
      const rect = canvas.getBoundingClientRect();
      const clientX = e.touches ? e.touches[0].clientX : e.clientX;
      const clientY = e.touches ? e.touches[0].clientY : e.clientY;
      return {
        x: clientX - rect.left,
        y: clientY - rect.top
      };
    }

    function startDrawing(e) {
      e.preventDefault();
      dibujando = true;
      const pos = getPos(e);
      ctx.beginPath();
      ctx.moveTo(pos.x, pos.y);
    }

    function draw(e) {
      if (!dibujando) return;
      e.preventDefault();
      const pos = getPos(e);
      ctx.lineTo(pos.x, pos.y);
      ctx.stroke();
      haDibujado = true;
    }

    function stopDrawing() {
      dibujando = false;
    }

    // Mouse Events
    canvas.addEventListener('mousedown', startDrawing);
    canvas.addEventListener('mousemove', draw);
    window.addEventListener('mouseup', stopDrawing);

    // Touch Events
    canvas.addEventListener('touchstart', startDrawing, { passive: false });
    canvas.addEventListener('touchmove', draw, { passive: false });
    canvas.addEventListener('touchend', stopDrawing);

    function limpiarCanvas() {
      ctx.clearRect(0, 0, canvas.width, canvas.height);
      haDibujado = false;
    }

    function validarFormulario() {
      const pin1 = document.getElementById('pin1').value;
      const pin2 = document.getElementById('pin2').value;

      if (pin1 !== pin2) {
        alert("Los PINs de seguridad no coinciden. Por favor verifícalos.");
        return false;
      }

      if (!haDibujado) {
        alert("Es obligatorio capturar su firma autógrafa en el recuadro indicado.");
        return false;
      }

      // Store image data in hidden field
      document.getElementById('firma_base64').value = canvas.toDataURL('image/png');

      // Simulate success for UI demo
      alert("Registro completado exitosamente. Su identidad digital ha sido vinculada.");
      return true;
    }
  </script>

        filFolio.addEventListener('keyup', renderTable);

        renderTable();
        renderCarpetasDominio();
    }

    function renderCarpetasDominio() {
        const container = document.getElementById('listCarpetasDominio');
        container.innerHTML = '';

        if (carpetasDominio.length === 0) {
            container.innerHTML = '<p class="text-[11px] text-slate-400 text-center py-4 italic">No hay carpetas configuradas.</p>';
            return;
        }

        carpetasDominio.forEach(c => {
            container.innerHTML += `
                <div class="bg-slate-50 p-3 rounded-xl border border-slate-100 relative group">
                    <strong class="block text-xs text-primary mb-1">@${c.dominio}</strong>
                    <span class="text-[10px] text-slate-500 break-all block pr-6">ID: ${c.drive_folder_id}</span>
                    <button onclick="eliminarCarpetaDominio(${c.id})" class="absolute top-3 right-3 text-red-400 hover:text-red-600">
                        <span class="material-symbols-outlined text-[16px]">close</span>
                    </button>
                </div>
            `;
        });
    }

    function guardarCarpetaDominio() {
        const dominio = document.getElementById('inpDominio').value;
        const drive_folder_id = document.getElementById('inpDriveDom').value;
        if(!dominio || !drive_folder_id) return alert("Completa ambos campos");

        ejecutarAccion('guardar_carpeta_dominio', {dominio, drive_folder_id}).then(() => {
            window.location.reload();
        });
    }

    function eliminarCarpetaDominio(id) {
        if(confirm("¿Seguro de eliminar esta configuración de carpeta para el dominio?")) {
            ejecutarAccion('eliminar_carpeta_dominio', {id}).then(() => window.location.reload());
        }
    }

    function renderTable() {
        let docs = [...rawData];

        const fFolio = filFolio.value.toLowerCase();
        const fDom = filDominio.value;
        const fIni = filFechaIni.value;
        const fFin = filFechaFin.value;

        if(fFolio) docs = docs.filter(d => d.reference_id.toLowerCase().includes(fFolio));
        if(fDom !== 'all') docs = docs.filter(d => d.dominio === fDom);
        if(fIni) docs = docs.filter(d => d.fecha.split(' ')[0] >= fIni);
        if(fFin) docs = docs.filter(d => d.fecha.split(' ')[0] <= fFin);

        tableBody.innerHTML = '';
        const groupProp = selGroup.value;

        if (groupProp === 'none') {
            docs.forEach(doc => tableBody.appendChild(crearFila(doc)));
        } else {
            const groups = {};
            docs.forEach(d => {
                const key = d[groupProp];
                if(!groups[key]) groups[key] = [];
                groups[key].push(d);
            });

            for (const [groupName, items] of Object.entries(groups)) {
                const trGroup = document.createElement('tr');
                trGroup.className = 'group-header';
                trGroup.innerHTML = `<td colspan="7" class="py-3 px-4 border-l-4 border-primary">Grupo: ${groupName} <span class="ml-2 font-normal text-slate-500">(${items.length} docs)</span></td>`;
                tableBody.appendChild(trGroup);

                items.forEach(doc => tableBody.appendChild(crearFila(doc)));
            }
        }
    }

    function crearFila(doc) {
        const tr = document.createElement('tr');
        tr.className = 'hover:bg-slate-50 transition-colors';

        let badgeClass = 'bg-processing'; let statusText = 'En Proceso';
        if(doc.status === 'COMPLETED') { badgeClass = 'bg-completed'; statusText = 'Completado'; }
        if(doc.status === 'CANCELLED') { badgeClass = 'bg-cancelled'; statusText = 'Cancelado'; }

        let cancelBtn = '';
        if(doc.status === 'PROCESSING') {
            cancelBtn = `<button class="text-red-500 hover:text-red-700 text-[11px] font-bold underline" onclick="cancelarDoc('${doc.token}')">Cancelar</button>`;
        }

        tr.innerHTML = `
            <td><span class="font-bold text-primary">${doc.reference_id}</span></td>
            <td class="text-on-surface-variant">${doc.owner_email}</td>
            <td class="text-on-surface-variant font-medium text-xs">@${doc.dominio}</td>
            <td class="text-on-surface-variant text-xs">${doc.fecha}</td>
            <td class="text-primary font-semibold text-xs">${doc.progreso} Firmas</td>
            <td><span class="badge ${badgeClass}">${statusText}</span></td>
            <td>
                <div class="flex items-center gap-4">
                    <a href="/trazabilidad/${doc.token}/" target="_blank" class="text-primary font-bold text-[11px] hover:underline flex items-center gap-1">
                        <span class="material-symbols-outlined text-[16px]">visibility</span> Ver
                    </a>
                    ${cancelBtn}
                </div>
            </td>
        `;
        return tr;
    }

    function cancelarDoc(token) {
        if(confirm("ATENCIÓN: Cancelar el documento bloqueará la recolección de firmas de forma permanente. ¿Deseas continuar?")) {
            ejecutarAccion('cancelar_doc', { token: token }).then(() => {
                const obj = rawData.find(d => d.token === token);
                if(obj) obj.status = 'CANCELLED';
                renderTable();
            });
        }
    }

    function eliminarPlantilla(id) {
        if(confirm("¿Estás seguro de eliminar esta plantilla de forma permanente?")) {
            ejecutarAccion('eliminar_plantilla', { id: id }).then(() => window.location.reload());
        }
    }

    function guardarDiseno() {
        const config = {
            group: selGroup.value,
            filFolio: filFolio.value,
            filDominio: filDominio.value,
            filFechaIni: filFechaIni.value,
            filFechaFin: filFechaFin.value
        };
        ejecutarAccion('guardar_config', { configuracion: config }).then(() => {
            alert("Configuración de vista guardada correctamente.");
        });
    }

    function ejecutarAccion(accion, payload) {
        return fetch(`/api/admin-action/${accion}/`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(payload)
        })
        .then(r => r.json())
        .then(data => {
            if(data.status === 'success') {
                if(data.msg) alert(data.msg);
                return true;
            }
            else { alert("Error: " + data.error); throw new Error(); }
        });
    }

    function filtrarPlantillas() {
      const input = document.getElementById('buscarPlantilla');
      const filter = input.value.toLowerCase().trim();
      const grid = document.getElementById('gridPlantillas');
      const cards = grid.getElementsByClassName('template-card');
      const emptyState = document.getElementById('emptyState');
      const countBadge = document.getElementById('countBadge');

      let visibleCount = 0;

      for (let i = 0; i < cards.length; i++) {
        const nombre = cards[i].getAttribute('data-nombre');
        if (nombre.indexOf(filter) > -1) {
          cards[i].classList.remove('hidden');
          visibleCount++;
        } else {
          cards[i].classList.add('hidden');
        }
      }

      // Update badge count
      countBadge.innerText = `${visibleCount} Plantilla${visibleCount !== 1 ? 's' : ''}`;

      // Toggle empty state
      if (visibleCount === 0) {
        grid.classList.add('hidden');
        emptyState.classList.remove('hidden');
      } else {
        grid.classList.remove('hidden');
        emptyState.classList.add('hidden');
      }
    }
  </script>

    // Force uppercase for specific fields while preserving cursor position

      if (e.target.tagName === 'INPUT') {

        const id = e.target.id;
        const excludedIds = ['p_drive', 'p_usuarios'];

          e.target.value = e.target.value.toUpperCase();
          try { e.target.setSelectionRange(start, end); } catch(err) {}
        }
      }

    function analizarDoc() {
      const doc_id = document.getElementById('doc_id').value;
      const owner = document.getElementById('owner').value;
      if(!doc_id || !owner) return alert("Por favor, complete los campos iniciales.");

      const btn = event.currentTarget;
      const originalText = btn.innerHTML;
      btn.innerHTML = `<span class="animate-pulse">Analizando con IA...</span>`;
      btn.disabled = true;

      // Mock API response based on provided business logic
      fetch('/api/admin-action/analizar_plantilla/', {
        method: 'POST', headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ doc_id: doc_id, owner_email: owner })
      }).then(r => r.json()).then(resp => {
        if(resp.status !== 'success') {
          alert("Error al analizar el documento.");
          btn.innerHTML = originalText;
          btn.disabled = false;
          return;
        }

        try {
          let dataExtraida = resp.data;
          if (Array.isArray(dataExtraida)) dataExtraida = dataExtraida[0];
          let textToParse = dataExtraida.content?.parts?.[0]?.text || dataExtraida.text || dataExtraida;

          if (typeof textToParse === 'string') {
            let cleanedJsonString = textToParse.replace(/```json/gi, '').replace(/```/g, '').trim();
            rawData = JSON.parse(cleanedJsonString);
          } else {
            rawData = dataExtraida;
          }

          document.getElementById('txtContexto').innerText = rawData.contexto || "Análisis no disponible";
          document.getElementById('txtIntencion').innerText = rawData.intencion || "Análisis no disponible";

          const tbV = document.getElementById('tbodyVars'); tbV.innerHTML = '';
          if(rawData.variables && Array.isArray(rawData.variables)) {
            rawData.variables.forEach(v => {
              if(!v.key.includes("FIRMA_")) {
                tbV.innerHTML += `
                  <tr class="hover:bg-gray-50 transition-colors">
                    <td class="px-4 py-3 font-medium text-gray-900">${v.label}</td>
                    <td class="px-4 py-3 text-gray-500 font-mono text-xs">${v.key}</td>
                    <td class="px-4 py-3"><input type="email" class="var-email w-full text-xs border-gray-200 rounded p-1" data-key="${v.key}" value="${owner}"></td>
                  </tr>`;
              }
            });
          }

          const tbF = document.getElementById('tbodyFirmas'); tbF.innerHTML = '';
          if(!rawData.firmas || rawData.firmas.length === 0) {
            rawData.firmas = ["{{FIRMA_1}}"];
          }

          rawData.firmas.forEach((f, idx) => {
            tbF.innerHTML += `

        } catch (err) {
          console.error("Error parseando respuesta:", err);
          alert("Error interpretando datos de IA. Ver consola.");

          btn.disabled = false;
        }
      }).catch(err => {

        btn.innerHTML = originalText;

    function habilitarDragAndDrop() {
      const tableBody = document.getElementById('tbodyFirmas');
      let draggedRow = null;

      tableBody.querySelectorAll('tr').forEach(row => {
        row.addEventListener('dragstart', function (e) {
          draggedRow = this;
          setTimeout(() => this.classList.add('dragging'), 0);
        });

        row.addEventListener('dragend', function () {
          this.classList.remove('dragging');
          actualizarOrdenFirmas();
        });

        row.addEventListener('dragover', e => e.preventDefault());

        row.addEventListener('drop', function (e) {
          e.preventDefault();
          if (draggedRow !== this) {
            const rows = Array.from(tableBody.children);
            const draggedIdx = rows.indexOf(draggedRow);
            const targetIdx = rows.indexOf(this);
            if (draggedIdx < targetIdx) tableBody.insertBefore(draggedRow, this.nextSibling);
            else tableBody.insertBefore(draggedRow, this);
          }
        });
      });
    }

    const fileInput = document.getElementById('pdfFile');
    const dropZone = document.getElementById('dropZone');
    const fileNameDisplay = document.getElementById('fileNameDisplay');
    const uploadBtn = document.getElementById('uploadBtn');

    // Handle Drag and Drop visual states
    ['dragenter', 'dragover'].forEach(eventName => {
      dropZone.addEventListener(eventName, (e) => {
        e.preventDefault();
        dropZone.classList.add('drag-over');
      }, false);
    });

    ['dragleave', 'drop'].forEach(eventName => {
      dropZone.addEventListener(eventName, (e) => {
        e.preventDefault();
        dropZone.classList.remove('drag-over');
      }, false);
    });

    // Handle file drop
    dropZone.addEventListener('drop', (e) => {
      const dt = e.dataTransfer;
      const files = dt.files;
      if (files.length) {
        fileInput.files = files;
        handleFileSelection();
      }
    });

    // Handle click to select
    dropZone.addEventListener('click', () => fileInput.click());

    // Handle input change
    fileInput.addEventListener('change', handleFileSelection);

    function handleFileSelection() {
      if (fileInput.files.length > 0) {
        const file = fileInput.files[0];
        if (file.type !== "application/pdf") {
          alert("Solo se permiten archivos en formato PDF.");
          fileInput.value = "";
          fileNameDisplay.innerText = "Arrastra tu PDF aquí o haz clic";
          return;
        }
        fileNameDisplay.innerText = file.name;
        fileNameDisplay.classList.add('text-brand-accent');
      }
    }

      const file = fileInput.files[0];
      uploadBtn.disabled = true;

          <svg class="animate-spin h-5 w-5 text-white" xmlns="http://www.w3.org/2000/svg" fill="none" viewBox="0 0 24 24">

          </svg>
          Subiendo archivo de forma segura...
        </span>
      `;

      const formData = new FormData();
      formData.append('pdf_file', file);

      // Simulation of actual API call
      fetch('/api/subir-pdf-usuario/', {
          method: 'POST',
          body: formData
      }).then(r => r.json()).then(res => {
          if(res.status === 'success') {
              document.getElementById('docName').innerText = res.nombre;

    /**
     * Handles the addition of a new administrator via the API.
     */
    function agregarAdmin() {
      const emailInput = document.getElementById('newAdminEmail');
      const email = emailInput.value;

      if (!email || !email.includes('@')) {
        alert("Por favor, ingresa un correo electrónico válido.");
        return;
      }

      // Simulation of API call based on baseline logic
      fetch('/api/admin-action/agregar_admin/', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ email: email })
      })
      .then(r => r.json())
      .then(res => {
        if (res.status === 'success') {
          location.reload();
        } else {
          alert("Error: " + (res.error || "No se pudo añadir al administrador"));
        }
      })
      .catch(err => {
        console.error("Fetch error:", err);
        // Note: In a real demo this will fail, but logic matches requirement.
      });
    }

    /**
     * Updates the superadmin privileges for a specific user.
     * @param {number} id - The internal ID of the admin.
     * @param {boolean} isSuper - The new status.
     */
    function toggleSuperAdmin(id, isSuper) {
      fetch('/api/admin-action/actualizar_admin/', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ id: id, es_superadmin: isSuper })
      })
      .then(r => r.json())
      .then(res => {
        if (res.status !== 'success') {
          alert("Error al actualizar privilegios: " + res.error);
          location.reload();
        }
      });
    }

    /**
     * Removes an administrator from the system.
     * @param {number} id - The internal ID of the admin.
     * @param {string} email - The email for display in the confirmation dialog.
     */
    function eliminarAdmin(id, email) {
      if (confirm(`¿Estás seguro de que deseas eliminar al administrador ${email}?\nEsta acción no se puede deshacer.`)) {
        fetch('/api/admin-action/eliminar_admin/', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ id: id })
        })
        .then(r => r.json())
        .then(res => {
          if (res.status === 'success') {
            location.reload();
          } else {
            alert("Error: " + res.error);
          }
        });
      }
    }

<script data-purpose="auth-logic">
    /**
     * Handles the login process by validating fields and sending a request to the server.
     */
    function iniciarSesion() {
      const email = document.getElementById('email').value;
      const pin = document.getElementById('pin').value;

      if (!email || !pin) {
        alert("Por favor, completa todos los campos requeridos.");
        return;
      }

      const btn = document.getElementById('btnLogin');
      const originalText = btn.innerText;

      btn.disabled = true;
      btn.innerHTML = `
        <svg class="animate-spin -ml-1 mr-3 h-5 w-5 text-white inline-block" xmlns="http://www.w3.org/2000/svg" fill="none" viewBox="0 0 24 24">
          <circle class="opacity-25" cx="12" cy="12" r="10" stroke="currentColor" stroke-width="4"></circle>
          <path class="opacity-75" fill="currentColor" d="4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4zm2 5.291A7.962 7.962 0 014 12H0c0 3.042 1.135 5.824 3 7.938l3-2.647z"></path>
        </svg>
        Verificando...
      `;

      // Mock of fetch call based on original code structure
      fetch('{% url "admin_login" %}', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ email: email, pin: pin })
      })
      .then(r => r.json())
      .then(data => {
        if (data.status === 'success') {
          window.location.href = "{% url 'admin_dashboard' %}";
        } else {
          alert(data.error || "Error de autenticación. Intente de nuevo.");
          btn.disabled = false;
          btn.innerText = originalText;
        }
      })
      .catch(err => {
        // Fallback for environment without backend or network error
        console.error("Auth Error:", err);
        btn.disabled = false;
        btn.innerText = originalText;
        alert("Error al conectar con el servidor.");
      });
    }

    /**
     * Requests a One-Time PIN to be sent to the user's email.
     */
    function pedirOTP() {
      const email = document.getElementById('email').value;
      if (!email) {
        alert("Ingresa tu correo administrativo para recibir el PIN.");
        document.getElementById('email').focus();
        return;
      }

      const btn = document.getElementById('btnOtp');
      const originalText = btn.innerHTML;
      btn.disabled = true;
      btn.innerText = "Solicitando...";

      fetch('{% url "solicitar_otp" %}', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ email: email })
      })
      .then(r => r.json())
      .then(data => {
        alert(data.msg || data.error || "Se ha enviado un PIN temporal a tu correo.");
        btn.innerText = "PIN Temporal Enviado";
        btn.classList.add('text-green-600');
        setTimeout(() => {
          btn.disabled = false;
          btn.innerHTML = originalText;
          btn.classList.remove('text-green-600');
        }, 5000);
      })
      .catch(err => {
        console.error("OTP Error:", err);
        btn.disabled = false;
        btn.innerHTML = originalText;
      });
    }

    // FORZAR MAYÚSCULAS REALES EN DATA
    document.addEventListener('input', function (e) {
        if (e.target.tagName === 'INPUT') {
            const type = e.target.type.toLowerCase();
            // Match inputs that should be uppercase
            if (e.target.classList.contains('uppercase-input')) {
                const start = e.target.selectionStart;
                const end = e.target.selectionEnd;
                e.target.value = e.target.value.toUpperCase();
                try { e.target.setSelectionRange(start, end); } catch(err) {}
            }
        }
    });

    function enviarSolicitud() {
        const btn = document.getElementById('submitBtn');
        const btnText = document.getElementById('btnText');

        btn.disabled = true;
        btnText.innerText = "Procesando envío...";

        // Collect Variables
        let docVars = {};
        document.querySelectorAll('.var-map').forEach(inp => {
          docVars[inp.dataset.key] = inp.value;
        });

        // Collect Signers
        let firmantesArr = [];
        const nombres = document.querySelectorAll('.firma-nombre');
        const correos = document.querySelectorAll('.firma-email');

        for(let i=0; i < nombres.length; i++) {
            if(nombres[i].value && correos[i].value) {
                firmantesArr.push({
                  nombre: nombres[i].value,
                  email: correos[i].value
                });
            }
        }

        // Validate basic fields
        if(firmantesArr.length === 0) {
          alert("Por favor rellena los datos de los firmantes.");
          btn.disabled = false;
          btnText.innerText = "Enviar Solicitud a Firma";
          return;
        }

        let folio = document.getElementById('folio_id').value;
        if(!folio) folio = "DOC-" + Math.floor(Date.now() / 1000);

        const payload = {
            reference_id: folio,
            dir: "{{ plantilla.carpeta_firmados_id|default:plantilla.drive_folder_id }}",
            template_id: "{{ plantilla.doc_id }}",
            view_info: "{{ plantilla.view_info }}",
            exec: "form",
            owner: "{{ owner_email }}",
            variables_asignadas: docVars,
            firmantes: firmantesArr
        };

        fetch('https://n8n.raloy.com.mx/webhook/request-signature', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(payload)
        }).then(r => {
            if(r.ok) {
                alert("¡Documento generado y enviado con éxito!");
                window.location.href = '/portal/dashboard/';
            } else {
                throw new Error("Error en el servidor");
            }
        }).catch(err => {
            alert("Error al enviar la solicitud. Intente nuevamente.");
            btn.disabled = false;
            btnText.innerText = "Enviar Solicitud a Firma";
        });
    }
  </script>

    const url = '{{ pdf_url }}'; // Injected by Django context
    const wrapper = document.getElementById('pdf-wrapper');

    let currentPage = 1;

    // 1. RENDER PDF
    pdfjsLib.getDocument(url).promise.then(pdfDoc => {
      pdfDocument = pdfDoc;
      totalPages = pdfDoc.numPages;
      renderPage(currentPage);
    }).catch(err => {
      console.error("Error loading PDF: ", err);
      // Fallback for demo purposes if URL is placeholder
      const canvas = document.getElementById('pdf-render');
      canvas.width = 600;
      canvas.height = 800;
      const ctx = canvas.getContext('2d');
      ctx.fillStyle = "#ffffff";
      ctx.fillRect(0,0,600,800);
      ctx.fillStyle = "#cbd5e1";
      ctx.font = "20px sans-serif";
      ctx.fillText("Visualización previa del documento", 150, 400);
      wrapper.style.width = '600px';
      wrapper.style.height = '800px';
    });

    function renderPage(num) {
      pdfDocument.getPage(num).then(page => {
        const scale = 1.5;
        const viewport = page.getViewport({ scale: scale });
        const canvas = document.getElementById('pdf-render');
        const ctx = canvas.getContext('2d');
        canvas.height = viewport.height;
        canvas.width = viewport.width;

        wrapper.style.width = viewport.width + 'px';
        wrapper.style.height = viewport.height + 'px';

        page.render({ canvasContext: ctx, viewport: viewport });
        document.getElementById('pageInfo').innerText = `Página ${num} de ${totalPages}`;

        // Show/Hide signatures based on current page
        boxes.forEach(b => {
          const el = document.getElementById(b.id);
          if(el) {
            el.style.display = (b.page === num) ? 'flex' : 'none';
          }
        });
      });
    }

    function changePage(delta) {
      let newPage = currentPage + delta;
      if(newPage >= 1 && newPage <= totalPages) {
        currentPage = newPage;
        renderPage(currentPage);
      }
    }

    // 2. DRAG & DROP LOGIC
    function agregarFirmante() {
      const id = 'sig-' + Date.now().toString();
      const ordenNum = boxes.length + 1;

      const box = document.createElement('div');
      box.className = 'signature-box';
      box.id = id;
      box.innerHTML = `<span class="sig-label">FIRMANTE ${ordenNum}</span><span class="text-[10px] opacity-60">Arrastrar aquí</span>`;
      box.style.left = '50px';
      box.style.top = '100px';

      const boxData = { id: id, orden: ordenNum, nombre: '', email: '', iniciales: '', page: currentPage };
      boxes.push(boxData);

      let isDragging = false, startX, startY, initialLeft, initialTop;

      box.addEventListener('mousedown', (e) => {
        seleccionarCaja(id);
        isDragging = true;
        startX = e.clientX; startY = e.clientY;
        initialLeft = parseInt(box.style.left || 0);
        initialTop = parseInt(box.style.top || 0);
        box.style.cursor = 'grabbing';
        e.stopPropagation();
      });

      document.addEventListener('mousemove', (e) => {
        if (!isDragging) return;
        let newX = initialLeft + (e.clientX - startX);
        let newY = initialTop + (e.clientY - startY);

        // Bounds checking
        if(newX < 0) newX = 0; if(newX > wrapper.clientWidth - 160) newX = wrapper.clientWidth - 160;
        if(newY < 0) newY = 0; if(newY > wrapper.clientHeight - 64) newY = wrapper.clientHeight - 64;

        box.style.left = newX + 'px';
        box.style.top = newY + 'px';
      });

      document.addEventListener('mouseup', () => {
        if(isDragging) { isDragging = false; box.style.cursor = 'grab'; }
      });

      wrapper.appendChild(box);
      seleccionarCaja(id);
    }

    // 3. UI SYNC
    function seleccionarCaja(id) {
      activeBoxId = id;
      document.querySelectorAll('.signature-box').forEach(b => b.classList.remove('selected'));
      const activeEl = document.getElementById(id);
      if(activeEl) activeEl.classList.add('selected');

      const data = boxes.find(b => b.id === id);
      if(data) {
        document.getElementById('propsPanel').classList.remove('hidden');
        document.getElementById('noSelectionMsg').classList.add('hidden');
        document.getElementById('inpOrden').value = data.orden;
        document.getElementById('inpNombre').value = data.nombre;
        document.getElementById('inpIniciales').value = data.iniciales;
        document.getElementById('inpEmail').value = data.email;
      }
    }

    function updateBoxData() {
      if(!activeBoxId) return;
      const data = boxes.find(b => b.id === activeBoxId);
      if(data) {
        data.orden = document.getElementById('inpOrden').value;
        data.nombre = document.getElementById('inpNombre').value.toUpperCase();
        data.iniciales = document.getElementById('inpIniciales').value.toUpperCase();
        data.email = document.getElementById('inpEmail').value;

        const label = document.getElementById(activeBoxId).querySelector('.sig-label');
        label.innerText = (data.nombre || 'FIRMANTE') + ' (' + data.orden + ')';
      }
    }

    function eliminarCajaActual() {

      boxes = boxes.filter(b => b.id !== activeBoxId);
      document.getElementById('propsPanel').classList.add('hidden');
      document.getElementById('noSelectionMsg').classList.remove('hidden');
      activeBoxId = null;
    }

    function enviarAFirma() {
      if(boxes.length === 0) return alert("Por favor, añada al menos un espacio de firma al documento.");

      let payloadFirmantes = [];
      let isValid = true;

      boxes.forEach(b => {
        if(!b.nombre || !b.email || !b.orden) isValid = false;

        const el = document.getElementById(b.id);
        const xPercent = parseInt(el.style.left) / wrapper.clientWidth;
        const yPercent = parseInt(el.style.top) / wrapper.clientHeight;

        payloadFirmantes.push({

          email: b.email,
          iniciales: b.iniciales,
          orden: parseInt(b.orden),
          coordenadas: { page: b.page, x: xPercent, y: yPercent }
        });
      });

      if(!isValid) return alert("Asegúrese de completar el nombre, correo y orden de todos los firmantes.");

      payloadFirmantes.sort((a, b) => a.orden - b.orden);

      const btn = document.getElementById('btnEnviar');
      const originalText = btn.innerHTML;
      btn.disabled = true;
      btn.innerHTML = `<svg class="animate-spin -ml-1 mr-3 h-5 w-5 text-white" xmlns="http://www.w3.org/2000/svg" fill="none" viewBox="0 0 24 24"><circle class="opacity-25" cx="12" cy="12" r="10" stroke="currentColor" stroke-width="4"></circle><path class="opacity-75" fill="currentColor" d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4zm2 5.291A7.962 7.962 0 014 12H0c0 3.042 1.135 5.824 3 7.938l3-2.647z"></path></svg> Procesando...`;

      fetch('/api/iniciar-firma-libre/', {

                return `${valor} Firmas Requeridas`;

            return valor;

        async function sendApiRequest() {
            const btn = document.getElementById('apiBtnSend');
            const responseBox = document.getElementById('apiResponse');
            let payloadStr = document.getElementById('apiPayload').value;
            let payloadObj;

            try {
                payloadObj = JSON.parse(payloadStr);
            } catch(e) {
                responseBox.innerText = "Error: El JSON no es válido.\n\n" + e.message;
                responseBox.style.color = '#b91c1c';
                responseBox.style.backgroundColor = '#fee2e2';
                return;
            }

            btn.disabled = true;
            btn.innerHTML = '<span class="material-symbols-outlined animate-spin">sync</span> Enviando...';

            responseBox.style.backgroundColor = '#f8fafc';

            try {
                const res = await fetch('https://n8n.raloy.com.mx/webhook/request-signature', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(payloadObj)
                });

                const text = await res.text();
                let jsonRes;
                try {
                    jsonRes = JSON.parse(text);
                    responseBox.innerText = JSON.stringify(jsonRes, null, 2);
                } catch(e) {
                    responseBox.innerText = text;
